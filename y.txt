        function fmt(n) {
            if (n === null || n === undefined || isNaN(n)) return '?';
            if (n >= 1_000_000) return (n / 1_000_000).toFixed(1) + 'M';
            if (n >= 1_000) return (n / 1_000).toFixed(1) + 'K';
            return n.toString();
        }

        async function fetchModrinthProject(slug) {
            const res = await fetch(`https://api.modrinth.com/v2/project/${slug}`);
            if (!res.ok) return { downloads: 0, icon_url: null };
            const data = await res.json();
            return { downloads: data.downloads || 0, icon_url: data.icon_url || null };
        }

        async function fetchCurseForgeDownloads(slug) {
            const res = await fetch(`https://api.cfwidget.com/minecraft/mc-mods/${slug}`);
            if (!res.ok) return 0;
            const data = await res.json();
            return data.downloads?.total ?? data.downloadCount ?? 0;
        }

        async function fetchModrinthUserTotal(username) {
            const r = await fetch(`https://api.modrinth.com/v2/user/${username}/projects`);
            if (!r.ok) return 0;
            const projects = await r.json();
            return projects.reduce((sum, p) => sum + (p.downloads || 0), 0);
        }
        async function fetchCFWidgetAll(slugs) {
            const results = await Promise.allSettled(slugs.map(s => fetchCurseForgeDownloads(s)));
            return results.reduce((sum, r) => r.status === 'fulfilled' ? sum + r.value : sum, 0);
        }

            fetchModrinthUserTotal(MR_USERNAME).catch(() => 0),
